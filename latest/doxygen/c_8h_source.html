222VSXML$(document)* VSXML$(parse)(const char* xml, size_t len, VSXML$(error_t)* err){
223 if(xml == nullptr){ set_err(err, VSXML$(ErrorInvalidArgument)); return nullptr; }
224 std::string buffer(xml, len); // mutable: the parser may unescape in place.
225 DocumentBuilder<builder_config_t{.symbols=builder_config_t::COMPRESS_ALL}> bld;
226 Parser parser(std::span<char>(buffer.data(), buffer.size()), bld);
227 if(auto r = parser.parse(); !r.has_value()){
228 set_err(err, VSXML$(ErrorParse));
229 return nullptr;
230 }
231 auto doc = bld.close();
232 if(!doc.has_value()){
233 set_err(err, VSXML$(ErrorGeneric));
234 return nullptr;
235 }
236 set_err(err, VSXML$(Ok));
237 return wrap_owned(std::move(*doc), err);
