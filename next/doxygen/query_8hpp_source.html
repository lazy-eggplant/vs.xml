155 for(size_t i=idx;i<steps.size();i++){
156 const step_t& s = steps[i];
157 switch(s.code){
158 case op::ACCEPT:

160 case op::CHILD:
161 if(node.type()==type_t::ELEMENT)
162 for(auto& c : node.children()) if(!run(c, i+1, fn))return false;

164 case op::DESCEND:
